X = birthrates["Year"]
y = birthrates["Birthrate"]

def fit_orthopoly(x, y, degree):
    X_poly = np.polynomial.legendre.legvander(
        np.interp(x, (x.min(), x.max()), (-1, 1)), degree
    )
    model = sm.OLS(y, X_poly).fit()
    
    x_grid = np.linspace(x.min(), x.max(), 300)
    x_scaled = np.interp(x_grid, (x.min(), x.max()), (-1, 1))
    X_grid_poly = np.polynomial.legendre.legvander(x_scaled, degree)
    y_pred = model.predict(X_grid_poly)
    return x_grid, y_pred

x3, y3 = fit_orthopoly(X, y, 3)

x6, y6 = fit_orthopoly(X, y, 6)

fig, axes = plt.subplots(1, 2, figsize=(12, 5))

axes[0].scatter(X, y, color="blue", s=20)
axes[0].plot(x3, y3, color="magenta", linewidth=2)
axes[0].set_title("degree = 3")

axes[1].scatter(X, y, color="blue", s=20)
axes[1].plot(x6, y6, color="magenta", linewidth=2)
axes[1].set_title("degree = 6")

plt.tight_layout()
plt.show()

We can see that the 6th order polynomial is better, but it is not very good. We can try fitting local polynomials - piecewise polynomials of lower order (we try constant functions and linear function in the example below:

years = birthrates["Year"].values
rates = birthrates["Birthrate"].values

myknots = [1936, 1960, 1978]
bounds = [1917] + myknots + [2003]

X_const = np.column_stack([
    (years < myknots[0]),
    (years >= myknots[0]) & (years < myknots[1]),
    (years >= myknots[1]) & (years < myknots[2]),
    (years >= myknots[2])
]).astype(int)

const_model = sm.OLS(rates, X_const).fit()

plt.figure(figsize=(6,4))
plt.scatter(years, rates, color="blue", s=20)
for knot in myknots:
    plt.axvline(knot, linestyle="--", color="black")

for k in range(4):
    plt.plot(
        [bounds[k], bounds[k+1]],
        [const_model.params[k]]*2,
        color="magenta", linewidth=4
    )

plt.title("Piecewise constant")
plt.show()

X_linear = np.column_stack([
    (years < myknots[0]),
    (years >= myknots[0]) & (years < myknots[1]),
    (years >= myknots[1]) & (years < myknots[2]),
    (years >= myknots[2]),
    years*(years < myknots[0]),
    years*((years >= myknots[0]) & (years < myknots[1])),
    years*((years >= myknots[1]) & (years < myknots[2])),
    years*(years >= myknots[2])
]).astype(float)

line_model = sm.OLS(rates, X_linear).fit()

plt.figure(figsize=(6,4))
plt.scatter(years, rates, color="blue", s=20)
for knot in myknots:
    plt.axvline(knot, linestyle="--", color="black")

for k in range(4):
    plt.plot(
        [bounds[k], bounds[k+1]],
        line_model.params[k] + np.array([bounds[k], bounds[k+1]]) * line_model.params[k+4],
        color="magenta", linewidth=4
    )

plt.title("Piecewise linear")
plt.show()

As we discussed in the lecture, it is a localized fit, but there are discontinuities and the overall fit is not good.

Splines Regression¶

Instead, we can use splines to fit the data. Let us first try to use the B-Splines. We will first try a linear spline

In [12]:
X_bs = dmatrix("bs(year, knots=(1936,1960,1978), degree=1, include_intercept=True)",
               {"year": birthrates["Year"]}, return_type='dataframe')

model_bs = sm.OLS(birthrates["Birthrate"], X_bs).fit()

fitted = model_bs.fittedvalues

plt.figure(figsize=(6,4))
plt.scatter(birthrates["Year"], birthrates["Birthrate"], color="blue", s=20)
plt.plot(birthrates["Year"], fitted, color="magenta", linewidth=4)
plt.show()

and then a cubic spline (as the ones discussed in class)
